# Regex rakendub kommentaarideta valijateksti algusele (nt '^\.login', '^:root', '^@keyframes login-').
import re, sys

def spans(text):
    out, i, depth, start = [], 0, 0, 0
    while i < len(text):
        if text.startswith('/*', i):
            i = text.index('*/', i) + 2
            continue
        c = text[i]
        if c == '{':
            depth += 1
        elif c == '}':
            depth -= 1
            if depth == 0:
                out.append((start, i + 1)); start = i + 1
        i += 1
    return out

def selector(chunk):
    return re.sub(r'\s+', ' ', re.sub(r'/\*.*?\*/', '', chunk, flags=re.S).split('{', 1)[0]).strip()

cmd, path = sys.argv[1], sys.argv[2]
text = open(path, encoding='utf-8').read()
sp = spans(text)
if cmd == 'list':
    for n, (a, b) in enumerate(sp, 1):
        print(n, text[a:b].count('\n') + 1, selector(text[a:b]))
    sys.exit(0)
rx = re.compile(sys.argv[3])
keep, taken, pos = [], [], 0
for a, b in sp:
    if rx.search(selector(text[a:b])):
        keep.append(text[pos:a]); taken.append(text[a:b].strip('\n')); pos = b
keep.append(text[pos:])
open(path, 'w', encoding='utf-8').write(re.sub(r'\n{3,}', '\n\n', ''.join(keep)).strip('\n') + '\n')
print('\n\n'.join(taken))
print(f'-- eemaldatud {len(taken)} plokki failist {path}', file=sys.stderr)

// Vana `@media (max-width: N)` = Tailwindi `max-[N+1px]` (max-[…] on range-süntaks: laius < N); sama konteineripäringul `@max-[…]`.
// `min-[Npx]` nihet ei vaja. Vana murdepunkt või ümmargune arv ilma +1-ta on peaaegu kindlasti viga.
const MAX_BREAKPOINT = /(?<![\w-])(@?max-\[(\d+)px\]):/g;
export const OLD_BREAKPOINTS: ReadonlySet<number> = new Set([520, 560, 640, 680, 720, 760, 860, 1100, 1400, 1655]);

/** Murdepunktid, kus +1 puudub: N jagub 10-ga või on tuntud vana max-width. */
export function offByOneBreakpoints(src: string): string[] {

    .filter((m) => Number(m[2]) % 10 === 0 || OLD_BREAKPOINTS.has(Number(m[2])))
    .map((m) => m[1]);
}

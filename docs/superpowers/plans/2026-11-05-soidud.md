```ts
// src/server/tripSegments.ts — telemeetria punktid → sõidud (süüde sisse → välja) ja nende mõõdikud.
// Puhas: andmebaasi ei puutu. Sõit jätkub üle lühikese pausi (süüde väljas ≤ PAUSE_MAX_MS);
// pikem väljalülitus või andmeauk > TRIP_GAP_MS lõpetab sõidu.
import { TRIP_GAP_MS } from './tripRoute.ts';

export type TripSample = {
  time: Date;
  lat: number;
  lng: number;
  ignition: boolean | null; // null = seade süütet ei saada → loetakse sees olevaks
  speedKmh: number | null;
  fuelUsedL: number | null; // kumulatiivne kütusekulu (l)
  fuelRateLph: number | null; // hetkekulu (l/h)
  fuelLevelPct: number | null;
  engineHours: number | null; // mootori koguaja näit (h)
};

export type SegmentedTrip = {
  startedAt: Date;
  endedAt: Date;
  status: 'ACTIVE' | 'CLOSED';
  points: TripSample[]; // ainult süüte sees punktid
  distanceKm: number;
  movingMinutes: number;
  idleMinutes: number;
  pauseCount: number;
  pauseMinutes: number;
  maxSpeedKmh: number | null;
  avgSpeedKmh: number | null;
  fuelLiters: number | null;
  fuelLevelStartPct: number | null;
  fuelLevelEndPct: number | null;
  engineHoursStart: number | null;
  engineHoursEnd: number | null;
  minLat: number;
  minLng: number;
  maxLat: number;
  maxLng: number;
  pointCount: number;
};

export const PAUSE_MAX_MS = 10 * 60 * 1000;
const STEP_MAX_MS = 5 * 60 * 1000; // pikem vahe süütega punktide vahel ei lähe liikumis-/seisuaega
const MIN_TRIP_MS = 60 * 1000;
const MIN_POINTS = 3;
const MOVING_KMH = 2;
const JITTER_M = 5;
const EARTH_RADIUS_M = 6371000;

const isOn = (s: TripSample) => s.ignition !== false;
const round = (n: number, d: number) => Math.round(n * 10 ** d) / 10 ** d;

function distanceM(a: TripSample, b: TripSample): number {
  const toRad = Math.PI / 180;
  const dLat = (b.lat - a.lat) * toRad;
  const dLng = (b.lng - a.lng) * toRad;
  const h = Math.sin(dLat / 2) ** 2 + Math.cos(a.lat * toRad) * Math.cos(b.lat * toRad) * Math.sin(dLng / 2) ** 2;
  return 2 * EARTH_RADIUS_M * Math.asin(Math.sqrt(h));
}

type Draft = { points: TripSample[]; breaks: Set<number>; pauseCount: number; pauseMs: number };

const firstOf = (pts: TripSample[], f: (p: TripSample) => number | null) => {
  for (const p of pts) { const v = f(p); if (v != null) return v; }
  return null;
};
const lastOf = (pts: TripSample[], f: (p: TripSample) => number | null) => {
  for (let i = pts.length - 1; i >= 0; i--) { const v = f(pts[i]); if (v != null) return v; }
  return null;
};

function fuelLiters(d: Draft): number | null {
  const counter = d.points.filter((p) => p.fuelUsedL != null);
  if (counter.length >= 2) {
    let sum = 0;
    for (let i = 1; i < counter.length; i++) {
      const inc = counter[i].fuelUsedL! - counter[i - 1].fuelUsedL!;
      if (inc > 0) sum += inc; // loenduri nullimine (langus) ei loe
    }
    return round(sum, 2);
  }
  let seen = false;
  let sum = 0;
  for (let i = 1; i < d.points.length; i++) {
    const prev = d.points[i - 1];
    if (prev.fuelRateLph == null) continue;
    seen = true;
    if (d.breaks.has(i)) continue;
    sum += prev.fuelRateLph * ((d.points[i].time.getTime() - prev.time.getTime()) / 3_600_000);
  }
  return seen ? round(sum, 2) : null;
}

function build(d: Draft, status: 'ACTIVE' | 'CLOSED'): SegmentedTrip {
  const pts = d.points;
  let distM = 0;
  let movingMs = 0;
  let idleMs = 0;
  let maxSpeed: number | null = null;
  let minLat = pts[0].lat, maxLat = pts[0].lat, minLng = pts[0].lng, maxLng = pts[0].lng;
  for (let i = 0; i < pts.length; i++) {
    const p = pts[i];
    if (p.speedKmh != null) maxSpeed = maxSpeed == null ? p.speedKmh : Math.max(maxSpeed, p.speedKmh);
    minLat = Math.min(minLat, p.lat); maxLat = Math.max(maxLat, p.lat);
    minLng = Math.min(minLng, p.lng); maxLng = Math.max(maxLng, p.lng);
    if (i === 0) continue;
    const step = distanceM(pts[i - 1], p);
    if (!(step < JITTER_M && (p.speedKmh ?? 0) < MOVING_KMH)) distM += step;
    if (d.breaks.has(i)) continue;
    const dt = p.time.getTime() - pts[i - 1].time.getTime();
    if ((p.speedKmh ?? 0) >= MOVING_KMH) movingMs += dt;
    else idleMs += dt;
  }
  const distanceKm = round(distM / 1000, 2);
  const movingMinutes = round(movingMs / 60_000, 1);
  return {
    startedAt: pts[0].time,
    endedAt: pts[pts.length - 1].time,
    status,
    points: pts,
    distanceKm,
    movingMinutes,
    idleMinutes: round(idleMs / 60_000, 1),
    pauseCount: d.pauseCount,
    pauseMinutes: round(d.pauseMs / 60_000, 1),
    maxSpeedKmh: maxSpeed,
    avgSpeedKmh: movingMinutes > 0 ? round(distanceKm / (movingMinutes / 60), 1) : null,
    fuelLiters: fuelLiters(d),
    fuelLevelStartPct: firstOf(pts, (p) => p.fuelLevelPct),
    fuelLevelEndPct: lastOf(pts, (p) => p.fuelLevelPct),
    engineHoursStart: firstOf(pts, (p) => p.engineHours),
    engineHoursEnd: lastOf(pts, (p) => p.engineHours),
    minLat, minLng, maxLat, maxLng,
    pointCount: pts.length
  };
}

/** Punktid (ajas kasvavalt) → sõidud. `now` määrab, kas viimane sõit on veel käimas. */
export function segmentTrips(samples: TripSample[], now: Date): SegmentedTrip[] {
  const drafts: Draft[] = [];
  let cur: Draft | null = null;
  let offSeen = false; // pärast viimast süütega punkti tuli süüde väljas
  for (const s of samples) {
    if (!isOn(s)) {
      if (cur) offSeen = true;
      continue;
    }
    if (cur) {
      const last = cur.points[cur.points.length - 1];
      const diff = s.time.getTime() - last.time.getTime();
      if (offSeen ? diff <= PAUSE_MAX_MS : diff <= TRIP_GAP_MS) {
        if (offSeen) {
          cur.pauseCount++;
          cur.pauseMs += diff;
          cur.breaks.add(cur.points.length);
        } else if (diff > STEP_MAX_MS) {
          cur.breaks.add(cur.points.length);
        }
        cur.points.push(s);
        offSeen = false;
        continue;
      }
      drafts.push(cur);
    }
    cur = { points: [s], breaks: new Set(), pauseCount: 0, pauseMs: 0 };
    offSeen = false;
  }
  let lastOpen = false;
  if (cur) {
    drafts.push(cur);
    const age = now.getTime() - cur.points[cur.points.length - 1].time.getTime();
    lastOpen = offSeen ? age <= PAUSE_MAX_MS : age <= TRIP_GAP_MS;
  }
  const out: SegmentedTrip[] = [];
  drafts.forEach((d, i) => {
    const span = d.points[d.points.length - 1].time.getTime() - d.points[0].time.getTime();
    if (d.points.length < MIN_POINTS || span < MIN_TRIP_MS) return;
    out.push(build(d, i === drafts.length - 1 && lastOpen ? 'ACTIVE' : 'CLOSED'));
  });
  return out;
}

/** fuel_consumed_counted väärtuse ühik liitrites (Teltonika loendur 0,1 l sammuga). */

/**
 * Ühe seadme punktid (fromExclusive, toInclusive] ettevõtte omandiperioodide piires, ajas kasvavalt.
 * Punkt ilma asukohata jäetakse välja. Süüde: value_bool (numbrilised varuks), välja puudumisel null.
 */
export async function fetchTripSamples(imei: string, fromExclusive: Date, toInclusive: Date): Promise<TripSample[]> {
  if (!tsdb) return [];
  const segments = clampWindow((await ownershipWindows('IMEI', [imei])).get(imei) ?? [], fromExclusive, toInclusive);
  if (segments.length === 0) return [];
  const froms = segments.map((s) => s.from.toISOString());
  const tos = segments.map((s) => s.to.toISOString());
  const incl = segments.map((s) => s.toInclusive);
  const rows = await tsdb.$queryRaw<{
    time: Date; lat: number | null; lng: number | null; ign: boolean | null; speed: number | null;
    fuel_used: number | null; fuel_rate: number | null; fuel_level: number | null; engine_hours: number | null;
  }[]>`
    WITH pts AS (
      SELECT m.time, m.field, m.value_bool,
        COALESCE(m.value_double, m.value_int::double precision, m.value_bigint::double precision) AS v
      FROM measurements m
      JOIN unnest(${froms}::timestamptz[], ${tos}::timestamptz[], ${incl}::boolean[]) AS s(seg_from, seg_to, seg_incl)
        ON m.time >= s.seg_from AND (m.time < s.seg_to OR (s.seg_incl AND m.time = s.seg_to))
      WHERE m.device = ${imei}
        AND m.time > ${fromExclusive} AND m.time <= ${toInclusive}
        AND m.measurement = 'data'
        AND m.field = ANY(${TRIP_FIELDS}::text[])
    )
    SELECT time,
      max(v) FILTER (WHERE field = 'latitude') AS lat,
      max(v) FILTER (WHERE field = 'longitude') AS lng,
      bool_or(COALESCE(value_bool, v <> 0)) FILTER (WHERE field = 'ignition') AS ign,
      max(v) FILTER (WHERE field = 'speed') AS speed,
      max(v) FILTER (WHERE field = 'fuel_consumed_counted') AS fuel_used,
      max(v) FILTER (WHERE field = 'fuel_rate_lph') AS fuel_rate,
      max(v) FILTER (WHERE field = 'fuel_level_percent') AS fuel_level,
      max(v) FILTER (WHERE field = 'engine_total_hours_h') AS engine_hours
    FROM pts
    GROUP BY time
    HAVING bool_or(field = 'latitude') AND bool_or(field = 'longitude')
    ORDER BY time ASC
  `;
  const num = (v: unknown) => (v == null ? null : Number(v));
  const out: TripSample[] = [];
  for (const r of rows) {
    if (r.lat == null || r.lng == null) continue;
    const used = num(r.fuel_used);
    out.push({
      time: r.time, lat: Number(r.lat), lng: Number(r.lng), ignition: r.ign ?? null,
      speedKmh: num(r.speed), fuelUsedL: used == null ? null : used * FUEL_CONSUMED_COUNTED_L,
      fuelRateLph: num(r.fuel_rate), fuelLevelPct: num(r.fuel_level), engineHours: num(r.engine_hours)
    });
  }
  return out;
}

function routeOf(t: { points: TripSample[] }, places: Place[]): TripRoute | null {
  try {
    return buildRoute(t.points, places, DEFAULT_ROUTE_OPTIONS);
  } catch (err) {
    console.warn('trip import: route failed', err);
    return null;
  }
}

const placeJson = (stop: TripRoute['start'] | undefined) =>
  stop ? { place: stop.place, ...(stop.near ? { near: stop.near } : {}) } : { place: null };

function tripData(t: SegmentedTrip, route: TripRoute | null, now: Date) {
  const first = t.points[0];
  const last = t.points[t.points.length - 1];
  return {
    startedAt: t.startedAt, endedAt: t.endedAt, status: t.status,
    startLat: first.lat, startLng: first.lng, startPlace: placeJson(route?.start),
    endLat: last.lat, endLng: last.lng, endPlace: placeJson(route?.end),
    visits: (route?.visits ?? []) as object[],
    distanceKm: t.distanceKm, movingMinutes: t.movingMinutes, idleMinutes: t.idleMinutes,
    pauseCount: t.pauseCount, pauseMinutes: t.pauseMinutes,
    maxSpeedKmh: t.maxSpeedKmh, avgSpeedKmh: t.avgSpeedKmh,
    fuelLiters: t.fuelLiters, fuelLevelStartPct: t.fuelLevelStartPct, fuelLevelEndPct: t.fuelLevelEndPct,
    engineHoursStart: t.engineHoursStart, engineHoursEnd: t.engineHoursEnd,
    minLat: t.minLat, minLng: t.minLng, maxLat: t.maxLat, maxLng: t.maxLng,
    pointCount: t.pointCount, computedAt: now
  };
}

async function writeChunk(deviceId: string, from: Date, to: Date, trips: SegmentedTrip[], routes: (TripRoute | null)[], now: Date, finalizedUntil: Date) {
  await tenantTransaction(async (tx) => {
    const existing = await tx.vehicleTrip.findMany({
      where: { deviceId, startedAt: { gt: from, lte: to } },
      orderBy: { startedAt: 'asc' },
      select: { id: true, startedAt: true, endedAt: true }
    });
    const used = new Set<string>();
    const matchIds = trips.map((t) => {
      const m = existing.find((e) => !used.has(e.id) && e.startedAt <= t.endedAt && e.endedAt >= t.startedAt);
      if (m) used.add(m.id);
      return m?.id ?? null;
    });
    const stale = existing.filter((e) => !used.has(e.id)).map((e) => e.id);
    if (stale.length) await tx.vehicleTrip.deleteMany({ where: { id: { in: stale } } });
    for (let i = 0; i < trips.length; i++) {
      const data = tripData(trips[i], routes[i], now);
      const id = matchIds[i];
      if (id) await tx.vehicleTrip.update({ where: { id }, data });
      else await tx.vehicleTrip.create({ data: { deviceId, ...data } });
    }
    await tx.tripImportState.upsert({
      where: { deviceId },
      create: { deviceId, finalizedUntil, lastRunAt: now, lastError: null, lastErrorAt: null },
      update: { finalizedUntil, lastRunAt: now, lastError: null, lastErrorAt: null }
    });
  }, { timeout: 60_000 });
}

export async function importDeviceTrips(device: Device, deps: ImportDeps = defaultDeps): Promise<void> {
  const windows = (await ownershipWindows('IMEI', [device.imei])).get(device.imei) ?? [];
  if (windows.length === 0) return;
  const state = await prisma.tripImportState.findUnique({ where: { deviceId: device.id } });
  const placesFor = placesCache(deps);
  const now = deps.now();
  let from = state?.finalizedUntil ?? new Date(Math.min(...windows.map((w) => w.from.getTime())));
  for (;;) {
    const to = new Date(Math.min(from.getTime() + CHUNK_MS, now.getTime()));
    const samples = await deps.fetchSamples(device.imei, from, to);
    const trips = segmentTrips(samples, to);
    const routes: (TripRoute | null)[] = [];
    for (const t of trips) routes.push(routeOf(t, await placesFor(t.startedAt.getUTCFullYear())));
    const finalCut = to.getTime() - FINAL_AFTER_MS;
    const firstOpen = trips.find((t) => t.status === 'ACTIVE' || t.endedAt.getTime() > finalCut);
    let next = firstOpen ? Math.min(finalCut, firstOpen.startedAt.getTime() - 1) : finalCut;
    if (next < from.getTime()) next = from.getTime();
    await writeChunk(device.id, from, to, trips, routes, now, new Date(next));
    if (to.getTime() >= now.getTime()) return;
    from = new Date(next);
  }
}

// --- Kohtade ümberarvutus --------------------------------------------------

async function placesFingerprint(): Promise<string> {
  const [poi, fs] = await Promise.all([
    prisma.pointOfInterest.aggregate({ _count: { _all: true }, _max: { updatedAt: true } }),
    prisma.fieldSeason.aggregate({ _count: { _all: true }, _max: { updatedAt: true } })
  ]);
  return `${poi._count._all}:${poi._max.updatedAt?.toISOString() ?? ''}:${fs._count._all}:${fs._max.updatedAt?.toISOString() ?? ''}`;
}

type Box = { minLat: number; minLng: number; maxLat: number; maxLng: number };

async function changedPlaceBoxes(since: Date): Promise<Box[]> {
  const [pois, seasons] = await Promise.all([
    prisma.pointOfInterest.findMany({ where: { updatedAt: { gt: since } }, select: { latitude: true, longitude: true } }),
    prisma.fieldSeason.findMany({ where: { updatedAt: { gt: since } }, select: { geometry: true } })
  ]);
  const boxes: Box[] = pois.map((p) => ({ minLat: p.latitude, maxLat: p.latitude, minLng: p.longitude, maxLng: p.longitude }));
  for (const s of seasons) {
    const g = s.geometry as any;
    const coords: number[][] = g?.type === 'Polygon' ? g.coordinates.flat() : g?.type === 'MultiPolygon' ? g.coordinates.flat(2) : [];
    if (coords.length === 0) continue;
    const lats = coords.map((c) => c[1]);
    const lngs = coords.map((c) => c[0]);
    boxes.push({ minLat: Math.min(...lats), maxLat: Math.max(...lats), minLng: Math.min(...lngs), maxLng: Math.max(...lngs) });
  }
  return boxes;
}

const overlaps = (a: Box, b: Box) =>
  a.minLat - PLACE_MARGIN_DEG <= b.maxLat && a.maxLat + PLACE_MARGIN_DEG >= b.minLat &&
  a.minLng - PLACE_MARGIN_DEG <= b.maxLng && a.maxLng + PLACE_MARGIN_DEG >= b.minLng;

function referencedIds(t: { startPlace: unknown; endPlace: unknown; visits: unknown }): string[] {
  const ids: string[] = [];
  for (const p of [t.startPlace, t.endPlace] as any[]) {
    if (p?.place?.id) ids.push(p.place.id);
    if (p?.near?.id) ids.push(p.near.id);
  }
  for (const v of ((t.visits as any[]) ?? [])) if (v?.place?.id) ids.push(v.place.id);
  return ids;
}

export async function refreshTripPlaces(device: Device, deps: ImportDeps = defaultDeps): Promise<number> {
  const state = await prisma.tripImportState.findUnique({ where: { deviceId: device.id } });
  if (!state) return 0;
  const fp = await placesFingerprint();
  if (fp === state.placesFingerprint) return 0;
  const now = deps.now();
  let count = 0;
  if (state.placesCheckedAt && state.finalizedUntil) {
    const boxes = await changedPlaceBoxes(state.placesCheckedAt);
    const placesFor = placesCache(deps);
    const trips = await prisma.vehicleTrip.findMany({ where: { deviceId: device.id, startedAt: { lte: state.finalizedUntil } } });
    for (const t of trips) {
      const year = t.startedAt.getUTCFullYear();
      const places = await placesFor(year);
      const known = new Set(places.map((p) => p.id));
      const hit = boxes.some((b) => overlaps(t, b)) || referencedIds(t).some((id) => !known.has(id));
      if (!hit) continue;
      const samples = await deps.fetchSamples(device.imei, new Date(t.startedAt.getTime() - 1), t.endedAt);
      const points = samples.filter((s) => s.ignition !== false);
      if (points.length === 0) continue;
      const route = routeOf({ points }, places);
      if (!route) continue;
      await prisma.vehicleTrip.update({
        where: { id: t.id },
        data: { startPlace: placeJson(route.start), endPlace: placeJson(route.end), visits: route.visits as object[], computedAt: now }
      });
      count++;
    }
  }
  await prisma.tripImportState.update({ where: { deviceId: device.id }, data: { placesFingerprint: fp, placesCheckedAt: now } });
  return count;
}

export async function importTenantTrips(deps: ImportDeps = defaultDeps): Promise<{ devices: number; failed: number }> {
  const devices = await prisma.telemetryDevice.findMany({
    where: { connectivityType: 'CELLULAR', imei: { not: null } },
    select: { id: true, imei: true }
  });
  let failed = 0;
  for (const d of devices) {
    const device = { id: d.id, imei: d.imei! };
    try {
      await importDeviceTrips(device, deps);
      await refreshTripPlaces(device, deps);
    } catch (err) {
      failed++;
      const msg = (err instanceof Error ? err.message : String(err)).slice(0, MAX_ERROR_LEN);
      console.error(`trip import: device ${device.imei} failed:`, err);
      const now = deps.now();
      await prisma.tripImportState.upsert({
        where: { deviceId: device.id },
        create: { deviceId: device.id, lastRunAt: now, lastError: msg, lastErrorAt: now },
        update: { lastRunAt: now, lastError: msg, lastErrorAt: now }
      }).catch((e) => console.error('trip import: state write failed', e));
    }
  }
  return { devices: devices.length, failed };
}

// Sõidud tulevad taustaimpordist (tripImport.ts); masin/haakeriist liidetakse lugemisel seoste registrist sõidu alguse ajal.
import type { IncomingMessage, ServerResponse } from 'http';
import { prisma } from './db.ts';
import { json, parseBody } from './http.ts';
import { fetchTripSamples } from './telemetryAccess.ts';

const DEFAULT_DAYS = 7;
const MAX_NOTES = 1000;

type TripRow = Awaited<ReturnType<typeof prisma.vehicleTrip.findMany>>[number];

const parseIso = (v: string | null): Date | null => {
  if (!v) return null;
  const d = new Date(v);
  return Number.isNaN(d.getTime()) ? null : d;
};

const deviceLabelOf = (d: { model: string; imei: string | null }) => (d.imei ? `${d.model} (${d.imei.slice(-4)})` : d.model);

async function resolveTargets(trips: TripRow[]) {
  const deviceIds = [...new Set(trips.map((t) => t.deviceId))];
  const [devices, assignments] = await Promise.all([
    prisma.telemetryDevice.findMany({ where: { id: { in: deviceIds } }, select: { id: true, model: true, imei: true } }),
    prisma.deviceAssignment.findMany({
      where: { deviceId: { in: deviceIds } },
      include: { vehicle: { select: { id: true, name: true } }, implement: { select: { id: true, name: true } } },
      orderBy: { startedAt: 'desc' }
    })
  ]);
  const vehicleIds = [...new Set(assignments.map((a) => a.vehicleId).filter((x): x is string => !!x))];
  const implAssignments = vehicleIds.length
    ? await prisma.implementAssignment.findMany({
        where: { vehicleId: { in: vehicleIds } },
        include: { implement: { select: { id: true, name: true } } },
        orderBy: { startedAt: 'desc' }
      })
    : [];
  const covers = (a: { startedAt: Date; endedAt: Date | null }, t: Date) => a.startedAt <= t && (!a.endedAt || a.endedAt >= t);
  return (t: TripRow) => {
    const dev = devices.find((d) => d.id === t.deviceId);
    const a = assignments.find((x) => x.deviceId === t.deviceId && covers(x, t.startedAt));
    let implement = a?.implement ?? null;
    if (!implement && a?.vehicleId) {
      implement = implAssignments.find((x) => x.vehicleId === a.vehicleId && covers(x, t.startedAt))?.implement ?? null;
    }
    return {
      deviceLabel: dev ? deviceLabelOf(dev) : t.deviceId,
      vehicleId: a?.vehicle?.id ?? null,
      vehicleName: a?.vehicle?.name ?? null,
      implementId: implement?.id ?? null,
      implementName: implement?.name ?? null
    };
  };
}

function toDto(t: TripRow, target: ReturnType<Awaited<ReturnType<typeof resolveTargets>>>) {
  const sp = (t.startPlace ?? { place: null }) as any;
  const ep = (t.endPlace ?? { place: null }) as any;
  return {
    id: t.id, deviceId: t.deviceId, ...target,
    startedAt: t.startedAt.toISOString(), endedAt: t.endedAt.toISOString(), status: t.status,
    start: { lat: t.startLat, lng: t.startLng, place: sp.place ?? null, ...(sp.near ? { near: sp.near } : {}) },
    end: { lat: t.endLat, lng: t.endLng, place: ep.place ?? null, ...(ep.near ? { near: ep.near } : {}) },
    visits: (t.visits as any[]) ?? [],
    distanceKm: t.distanceKm, movingMinutes: t.movingMinutes, idleMinutes: t.idleMinutes,
    pauseCount: t.pauseCount, pauseMinutes: t.pauseMinutes,
    maxSpeedKmh: t.maxSpeedKmh, avgSpeedKmh: t.avgSpeedKmh, fuelLiters: t.fuelLiters,
    fuelLevelStartPct: t.fuelLevelStartPct, fuelLevelEndPct: t.fuelLevelEndPct,
    engineHoursStart: t.engineHoursStart, engineHoursEnd: t.engineHoursEnd,
    pointCount: t.pointCount, notes: t.notes
  };
}

export async function handleTrips(req: IncomingMessage, res: ServerResponse, pathname: string, method: string, url: URL): Promise<boolean> {
  if (pathname !== '/api/fleet/trips' && !pathname.startsWith('/api/fleet/trips/')) return false;

  if (pathname === '/api/fleet/trips' && method === 'GET') {
    const to = parseIso(url.searchParams.get('to')) ?? new Date();
    const from = parseIso(url.searchParams.get('from')) ?? new Date(to.getTime() - DEFAULT_DAYS * 86_400_000);
    const status = url.searchParams.get('status');
    const rows = await prisma.vehicleTrip.findMany({
      where: {
        startedAt: { gte: from, lt: to },
        ...(status === 'ACTIVE' || status === 'CLOSED' ? { status } : {})
      },
      orderBy: { startedAt: 'desc' }
    });
    const target = await resolveTargets(rows);
    let list = rows.map((t) => toDto(t, target(t)));
    const vehicleId = url.searchParams.get('vehicleId');
    if (vehicleId) list = list.filter((t) => t.vehicleId === vehicleId);
    if (url.searchParams.get('unknownPlace') === '1') list = list.filter((t) => !t.start.place || !t.end.place);
    json(res, list);
    return true;
  }

  if (pathname === '/api/fleet/trips/import-status' && method === 'GET') {
    const states = await prisma.tripImportState.findMany({ include: { device: { select: { model: true, imei: true } } } });
    const last = states.map((s) => s.lastRunAt).filter((x): x is Date => !!x).sort((a, b) => b.getTime() - a.getTime())[0];
    json(res, {
      lastRunAt: last ? last.toISOString() : null,
      devices: states.map((s) => ({
        deviceId: s.deviceId, deviceLabel: deviceLabelOf(s.device),
        lastError: s.lastError, lastErrorAt: s.lastErrorAt ? s.lastErrorAt.toISOString() : null
      }))
    });
    return true;
  }

  const trackMatch = /^\/api\/fleet\/trips\/([^/]+)\/track$/.exec(pathname);
  if (trackMatch && method === 'GET') {
    const trip = await prisma.vehicleTrip.findUnique({ where: { id: trackMatch[1] }, include: { device: { select: { imei: true } } } });
    if (!trip || !trip.device.imei) { json(res, { error: 'Sõitu ei leitud' }, 404); return true; }
    const samples = await fetchTripSamples(trip.device.imei, new Date(trip.startedAt.getTime() - 1), trip.endedAt);
    json(res, { points: samples.filter((s) => s.ignition !== false).map((s) => [s.lat, s.lng]) });
    return true;
  }

  const idMatch = /^\/api\/fleet\/trips\/([^/]+)$/.exec(pathname);
  if (idMatch && (method === 'PATCH' || method === 'PUT')) {
    const id = idMatch[1];
    const existing = await prisma.vehicleTrip.findUnique({ where: { id }, select: { id: true } });
    if (!existing) { json(res, { error: 'Sõitu ei leitud' }, 404); return true; }
    const body = await parseBody(req);
    const notes = String(body?.notes ?? '').trim().slice(0, MAX_NOTES) || null;
    const saved = await prisma.vehicleTrip.update({ where: { id }, data: { notes } });
    const target = await resolveTargets([saved]);
    json(res, toDto(saved, target(saved)));
    return true;
  }

  return false;
}

          </select>
        </FilterClear>
        <FilterClear active={!!status} onClear={() => setStatus('')}>
          <select className="form-input" value={status} onChange={(e) => setStatus(e.target.value as '' | 'ACTIVE' | 'CLOSED')}>
            <option value="">Kõik olekud</option>
            <option value="ACTIVE">Käimas</option>
            <option value="CLOSED">Lõppenud</option>
          </select>
        </FilterClear>
        <FilterClear active={unknownPlace} onClear={() => setUnknownPlace(false)}>
          <label className="checkbox-label">
            <input type="checkbox" checked={unknownPlace} onChange={(e) => setUnknownPlace(e.target.checked)} />
            Ainult tundmatu algus või lõpp
          </label>
        </FilterClear>
      </div>

      {error && <div className="auth-error-banner">{error}</div>}

      {!loading && trips.length === 0 ? (
        <div className="empty-state">Valitud perioodil sõite pole.</div>
      ) : (
        <div className="table-wrapper">
          <table className="data-table">
            <thead>
              <tr>
                <th>Kuupäev</th>
                <th>Masin (haakeriist)</th>
                <th>Algus–lõpp</th>
                <th className="num">Kestus</th>
                <th>Kust → kuhu</th>
                <th className="num">km</th>
                <th className="num">Kütus l</th>
                <th className="num">Mootoritunnid</th>
                <th>Olek</th>
              </tr>
            </thead>
            <tbody>
              {trips.map((t) => {
                const active = t.status === 'ACTIVE';
                const engine = t.engineHoursStart != null && t.engineHoursEnd != null ? t.engineHoursEnd - t.engineHoursStart : null;
                return (
                  <tr key={t.id} className="clickable-row" onClick={() => setOpen(t)}>
                    <td>{formatDateTime(t.startedAt).slice(0, 10)}</td>
                    <td>
                      {t.vehicleName ?? <span className="text-muted">{t.deviceLabel} — seos puudub</span>}
                      {t.implementName && <span className="text-muted"> ({t.implementName})</span>}
                    </td>
                    <td>{formatDateTime(t.startedAt).slice(11)}–{active ? 'käimas' : formatDateTime(t.endedAt).slice(11)}</td>
                    <td className="num">{formatDuration(t.startedAt, t.endedAt)}</td>
                    <td>{placeText(t.start)} → {active ? '…' : placeText(t.end)}</td>
                    <td className="num">{formatQty(t.distanceKm)}</td>
                    <td className="num">{t.fuelLiters == null ? '—' : formatQty(t.fuelLiters)}</td>
                    <td className="num">{engine == null ? '—' : formatQty(engine)}</td>
                    <td>{active ? <span className="status-badge status-working">käimas</span> : 'lõppenud'}</td>
                  </tr>
                );
              })}
            </tbody>
          </table>
        </div>
      )}

      {open && (
        <TripDialog
          trip={open}
          onClose={() => setOpen(null)}
          onSaved={(saved) => { setTrips((list) => list.map((x) => (x.id === saved.id ? saved : x))); setOpen(saved); }}
        />
      )}
    </div>
  );
};
```
